find_package(pybind11 CONFIG QUIET)
if(NOT pybind11_FOUND)
  execute_process(COMMAND "${Python3_EXECUTABLE}" -m pybind11 --cmakedir OUTPUT_VARIABLE _pybind_dir OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE _pybind_result)
  if(NOT _pybind_result EQUAL 0)
    message(FATAL_ERROR "G2-B requires pybind11 in ${Python3_EXECUTABLE}")
  endif()
  list(PREPEND CMAKE_PREFIX_PATH "${_pybind_dir}")
  find_package(pybind11 CONFIG REQUIRED)
endif()
